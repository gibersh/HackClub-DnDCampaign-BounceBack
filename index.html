Roll a d20 to exit Room 2

from js import document def check_roll(event): output = document.getElementById("output") roll_value = document.getElementById("roll_input").value try: num = int(roll_value) if 1 <= num <= 20: output.innerHTML = f"You rolled {num}. Exiting Room 2..." # Disable button after success document.getElementById("roll_button").disabled = True else: output.innerHTML = "Please enter a number between 1 and 20." except ValueError: output.innerHTML = "Invalid input. Please enter an integer." # Attach click event to button document.getElementById("roll_button").addEventListener("click", check_roll)
